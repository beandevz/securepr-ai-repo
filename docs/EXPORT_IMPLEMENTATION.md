## Scoring Breakdown

Each severity level has a weight:
- Critical: 10 points
- High: 5 points
- Medium: 2 points
- Low: 1 point

**Calculation:**
- 3 Critical × 10 = 30 pts

**Total Score:** 32 points

**Gate Decision:**
- ✅ PASS if: score < 15 AND no CRITICAL issues
- ❌ FAIL if: any CRITICAL exists OR score ≥ 15

**Result:** FAIL (Critical issues found)

---

## Detailed Findings

### 1. SQL Injection Vulnerability

**Severity:** `CRITICAL`
**File:** `login.py` (line 45)
**OWASP Category:** A03:2021 - Injection
**Confidence:** 95%

**Description:**
The code builds a SQL query by directly inserting user input...

**Risk:**
Attacker can dump your entire database or bypass authentication.

**Recommendation:**
Use parameterized queries. The database driver handles escaping automatically.

**Vulnerable Code:**
```python
query = f"SELECT * FROM users WHERE id='{user_id}'"
return conn.execute(query).fetchone()
```

**Safe Fix:**
```python
query = "SELECT * FROM users WHERE id = %s"
return conn.execute(query, (user_id,)).fetchone()
```

---

## Next Steps

1. Review each finding above
2. Apply recommended fixes
3. Re-run SecurePR AI scan
4. Merge when all critical issues are resolved

### Bulk Export

```python
@router.post('/bulk-export')
async def bulk_export(job_ids: List[str], format: str = 'json'):
    """Export multiple scan results as a ZIP file."""
    import zipfile
    from io import BytesIO
    
    zip_buffer = BytesIO()
    with zipfile.ZipFile(zip_buffer, 'w', zipfile.ZIP_DEFLATED) as zip_file:
        for job_id in job_ids:
            result = await get_job_result(job_id)
            
            if format == 'json':
                content = json.dumps(result, indent=2)
                filename = f'{job_id}.json'
            elif format == 'csv':
                content = generate_csv(result)
                filename = f'{job_id}.csv'
            elif format == 'markdown':
                content = generate_markdown(result)
                filename = f'{job_id}.md'
            
            zip_file.writestr(filename, content)
    
    zip_buffer.seek(0)
    return StreamingResponse(
        zip_buffer,
        media_type='application/zip',
        headers={
            'Content-Disposition': f'attachment; filename=securepr-bulk-export.zip'
        }
    )
```

# backend/app/webhooks/export.py
@app.on_event("scan_completed")
async def on_scan_complete(job: Job):
    # Auto-export to configured destinations
    if settings.auto_export_enabled:
        result = job.result
        
        # Export to S3
        if settings.export_to_s3:
            await upload_to_s3(generate_json(result), f'{job.id}.json')
        
        # Export to Slack
        if settings.export_to_slack:
            await post_to_slack(generate_markdown(result))

            await create_jira_ticket(result)

interface ExportHistory {
  id: string;
  job_id: string;
  format: string;
  exported_at: string;
  exported_by: string;
  file_size: number;
}

// Log export
function logExport(jobId: string, format: string) {
  fetch('/api/export/history', {
    method: 'POST',
    body: JSON.stringify({
      job_id: jobId,
      format,
      exported_at: new Date().toISOString()
    })
  });
}

// View history
function ViewExportHistory() {
  const [history, setHistory] = useState<ExportHistory[]>([]);
  
  useEffect(() => {
    fetch('/api/export/history')
      .then(r => r.json())
      .then(setHistory);
  }, []);
  
  return (
    <table>
      {history.map(h => (
        <tr key={h.id}>
          <td>{h.job_id}</td>
          <td>{h.format}</td>
          <td>{new Date(h.exported_at).toLocaleString()}</td>
          <td>{h.file_size} bytes</td>
        </tr>
      ))}
    </table>
  );
}

async function generateWeeklyReport() {
  const lastWeek = await fetch('/api/jobs?since=7d').then(r => r.json());
  
  const summary = {
    total_scans: lastWeek.length,
    failed_scans: lastWeek.filter(j => j.verdict === 'FAIL').length,
    critical_issues: lastWeek.reduce((sum, j) => 
      sum + j.findings.filter(f => f.severity === 'critical').length, 0
    )
  };
  
  exportAsHTML(summary);
}
